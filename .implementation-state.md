        animateNebulas(nebulas, Date.now());
        animateConstellations(constels, timescale);
        
        controls.update();
        composer.render();
    }
    animate();
})();

let audioCtx, humGain, osc1, osc2, filter;
function initAudio() {
    if (audioCtx) return;
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    osc1 = audioCtx.createOscillator(); osc1.type='sine'; osc1.frequency.value=55;
    osc2 = audioCtx.createOscillator(); osc2.type='triangle'; osc2.frequency.value=110;
    filter = audioCtx.createBiquadFilter(); filter.type='lowpass'; filter.frequency.value=400;
    humGain = audioCtx.createGain(); humGain.gain.value=0.0;
    osc1.connect(filter); osc2.connect(filter); filter.connect(humGain); humGain.connect(audioCtx.destination);
    osc1.start(); osc2.start();
    humGain.gain.linearRampToValueAtTime(0.02, audioCtx.currentTime+2.0);
}
// modulateSynth(p): si p null -> ramp a 55/110/400 vol 0.02. Si p: isGasGiant = p.radius>5, isStar = p.isStar, f1/f2/fFilt/vol segun. osc1.type= isGasGiant?'triangle':'sine', osc2.type= isStar?'sawtooth':'triangle'.
// playBeep(freq,type,dur,vol): osc+gain, exponentialRampToValueAtTime(0.001, t+duration).
// playWarp: osc sine 200Hz -> 30Hz en 3s, gain 0.15 -> 0.001.
